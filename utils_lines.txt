64:def wrap_inline_tex(text: str) -> str:
65:    """
66:    wrap bare TeX commands (e.g. \\dot{M}) in $...$ so MathJax/KaTeX can render them.
67:    """
68:    if not text:
69:        return text
70:
71:    # normalize a few common TeX shorthands from arXiv summaries
72:    text = text.replace("\\~", "\\sim ")
73:    text = re.sub(r"\}(?=[A-Za-z0-9])", "} ", text)
74:
75:    # ensure commands that require an argument actually receive one
76:    def _fix_missing_args(match: re.Match[str]) -> str:
77:        cmd = match.group("cmd")
78:        arg = match.group("arg").strip()
79:        return f"\\{cmd}{{{arg}}}"
80:
81:    text = _ARG_FIX_RE.sub(_fix_missing_args, text)
82:
83:    out: list[str] = []
84:    i = 0
85:    in_math = False
86:    length = len(text)
87:
88:    def _consume_braced(src: str, start: int) -> tuple[str, int]:
89:        """return balanced braced sequence starting at start (which should point to '{')."""
90:        depth = 0
91:        idx = start
92:        buf: list[str] = []
93:        while idx < length:
94:            ch = src[idx]
95:            buf.append(ch)
96:            if ch == "{":
97:                depth += 1
98:            elif ch == "}":
99:                depth -= 1
100:                if depth == 0:
101:                    idx += 1
102:                    break
103:            idx += 1
104:        return ("".join(buf), idx)
105:
106:    while i < length:
107:        ch = text[i]
108:        if ch == "$":
109:            in_math = not in_math
110:            out.append(ch)
111:            i += 1
112:            continue
113:
114:        if ch == "\\" and not in_math:
115:            j = i + 1
116:            while j < length and text[j].isalpha():
117:                j += 1
118:
119:            if j == i + 1:
120:                out.append(ch)
121:                i += 1
122:                continue
123:
124:            cmd = text[i:j]
125:            if cmd in {"\\n", "\\r"}:
126:                out.append(cmd)
127:                i = j
128:                continue
129:
130:            expr_parts = [cmd]
131:            idx = j
132:
133:            # attach balanced brace groups immediately following command
134:            while idx < length and text[idx] == "{":
135:                braced, idx = _consume_braced(text, idx)
136:                expr_parts.append(braced)
137:
138:            # attach bare identifiers if present (e.g., \alpha r, \tilde x)
139:            while idx < length and text[idx].isalnum():
140:                expr_parts.append(text[idx])
141:                idx += 1
142:
143:            # include trailing sub/superscripts
144:            while idx < length and text[idx] in {"_", "^"}:
145:                marker = text[idx]
146:                expr_parts.append(marker)
147:                idx += 1
148:                if idx < length and text[idx] == "{":
149:                    braced, idx = _consume_braced(text, idx)
150:                    expr_parts.append(braced)
151:                elif idx < length:
152:                    expr_parts.append(text[idx])
153:                    idx += 1
154:
155:            expr = "".join(expr_parts)
156:            out.append(f"${expr}$")
157:            i = idx
158:            continue
159:
160:        out.append(ch)
161:        i += 1
162:
163:    wrapped = "".join(out).replace("$ $", "$")
164:    return wrapped

305:    if idx >= len(expr) or expr[idx] != "{":

330:def latex_to_plain(expr: str) -> str:
331:    result = []
332:    i = 0
333:    length = len(expr)
334:
335:    text_commands = {
336:        "\\mathrm",
337:        "\\textrm",
338:        "\\text",
339:        "\\textbf",
340:        "\\textit",
341:        "\\operatorname",
342:        "\\mathbf",
343:        "\\mathcal",
344:        "\\mathbb",
345:        "\\mathtt",
346:        "\\mathit",
347:        "\\rm",
348:        "\\bf",
349:        "\\emph",
350:    }
351:
352:    while i < length:
353:        ch = expr[i]
354:        if ch == "\\":
355:            j = i + 1
356:            while j < length and expr[j].isalpha():
357:                j += 1
358:            cmd = expr[i:j]
359:            bare_cmd = cmd.strip("\\")
360:
361:            if cmd in LATEX_SYMBOLS:
362:                result.append(LATEX_SYMBOLS[cmd])
363:                i = j
364:                continue
365:            if bare_cmd in ACCENT_COMBINERS:
366:                content, next_idx = _read_group(expr, j)
367:                if content:
368:                    result.append(_apply_accent(bare_cmd, latex_to_plain(content)))
369:                    i = next_idx
370:                    continue
371:            if cmd == "\\frac":
372:                num, after_num = _read_group(expr, j)
373:                den, after_den = _read_group(expr, after_num)
374:                if num and den:
375:                    result.append(f"({latex_to_plain(num)})/({latex_to_plain(den)})")
376:                    i = after_den
377:                    continue
378:            if cmd == "\\sqrt":
379:                radicand, next_idx = _read_group(expr, j)
380:                if radicand:
381:                    result.append(f"√({latex_to_plain(radicand)})")
382:                    i = next_idx
383:                    continue
384:            if cmd in text_commands:
385:                content, next_idx = _read_group(expr, j)
386:                if content:
387:                    result.append(latex_to_plain(content))
388:                    i = next_idx
389:                    continue
390:            if cmd in {"\\left", "\\right"}:
391:                i = j
392:                continue
393:            # unrecognised command: drop backslash and keep text
394:            result.append(expr[i + 1 : j])
395:            i = j
396:            continue
397:        elif ch == "^":
398:            if i + 1 < length and expr[i + 1] == "{":
399:                content, next_idx = _read_group(expr, i + 1)
400:                result.append(_to_superscript(latex_to_plain(content)))
401:                i = next_idx
402:            elif i + 1 < length:
403:                result.append(_to_superscript(expr[i + 1]))
404:                i += 2
405:            else:
406:                i += 1
407:            continue
408:        elif ch == "_":
409:            if i + 1 < length and expr[i + 1] == "{":
410:                content, next_idx = _read_group(expr, i + 1)
411:                result.append(_to_subscript(latex_to_plain(content)))
412:                i = next_idx
413:            elif i + 1 < length:
414:                result.append(_to_subscript(expr[i + 1]))
415:                i += 2
416:            else:
417:                i += 1
418:            continue
419:        elif ch in "{}":
420:            i += 1
421:            continue
422:        else:
423:            result.append(ch)
424:            i += 1
425:
426:    return "".join(result)

430:    """
431:    Convert inline math ($...$) to KaTeX HTML spans suitable for email clients.
432:    Returns (processed_html, math_found).

446:        return match.group(0)

451:def fetch_arxiv_feed(url):
452:    """fetch and parse the arXiv API XML feed, returning a list of papers."""
453:    try:
454:        r = requests.get(url, timeout=10)
455:        r.raise_for_status()
456:        root = ET.fromstring(r.text)
457:        ns = {"atom": "http://www.w3.org/2005/Atom"}
458:
459:        entries = []
460:        for entry in root.findall("atom:entry", ns):
461:            title = entry.find("atom:title", ns)
462:            summary = entry.find("atom:summary", ns)
463:            link = entry.find("atom:id", ns)
464:
465:            summary_text = summary.text.strip() if summary is not None else ""
466:            entries.append(
467:                {
468:                    "title": title.text.strip() if title is not None else "(no title)",
469:                    "summary": wrap_inline_tex(summary_text),
470:                    "link": link.text.strip() if link is not None else "#",
471:                }
472:            )
473:
474:        print(f"[fetch_arxiv_feed] parsed {len(entries)} entries")
475:        return entries
476:    except Exception as e:
477:        print(f"[fetch_arxiv_feed] error: {e}")
478:        return []
